### **三、数据驱动建模流程**

#### **1.数据预处理**

**剔除共线性**：使用方差膨胀因子（VIF）筛选变量（如剔除周长，保留形状系数\(C_s=P^2/(4\piA)\)）。

**非线性变换**：对面积\(A\)、路径长度\(L\)取自然对数。

 

#### **2.多元回归模型**

[

ln(Q_p)=\beta_0+\beta_1\ln(A)+\beta_2J+\beta_3n+\beta_4f+\epsilon

]

参数估计：最小二乘法或岭回归（应对多重共线性）。

 

#### **3.机器学习模型**

**特征工程**：加入交互项（如\(J\timesL\)反映能量梯度）。

**超参数调优**：网格搜索确定树深度、学习率。

**输出**：特征重要性排序+SHAP值解释（如下图）。

def peak_modulus(P, F, A, B=0.7):
    Q = max(P - F, 0)
    return Q * A ** (-B)


def peak_modulus_stat(k, J, S, n, f, A):
    print(k, J, S, n, f, A)
    return k * J ** 0.5 * S ** 0.3 / (n ** 0.8 * f ** 0.4 * A ** 0.25)


# Rh （m，坡面流或渠道流的等效流深）
def confluence_time_wave(L, RH, J, n):
    return L * 1000 * n / (RH ** 0.666 * J ** 0.5)


def empirical_confluence_time(L, J):
    return 0.0195 * ((L * 1000) ** 0.77) / (J ** 0.385)


def peak_time_nash(tc, Ks, P, A):
    alpha = 0.6 if Ks < 1.5 else 0.9
    return alpha * tc * (1 + 0.1 * P / A ** 0.2)


def max_confluence_time(Tp, A, Qp, Kb=1):
    return Tp + Kb * A / Qp


def max_confluence_time_stat(A, Ks, n):
    return 1.2 * A ** 0.35 * Ks ** (-0.15) * n ** 0.25

    'k': fields.Float(required=True, description='经验系数 k（一般为 0.2~0.3）'),
    'J': fields.Float(required=True, description='比降 J（单位：m/m）'),
    'S': fields.Float(required=True, description='坡度 S（单位：度°）'),
    'n': fields.Float(required=True, description='曼宁糙率 n（无量纲 或s/m^1/3）'),
    'f': fields.Float(required=True, description='下渗率 f（单位：mm/h）'),
    'L': fields.Float(required=True, description='最长汇流路径 L（单位：km）'),
    'RH': fields.Float(required=True, description='水力半径 RH（单位：m）近似为0.1~0.3m'),

indicator_output = api.model('IndicatorResult', {

    'peak_time_nash': fields.Float(description='汇流峰值时间（Nash模型，单位：h）'),
    'max_confluence_time': fields.Float(description='汇流最长时间（线性水库模型，单位：h）'),

})


@ns.route('/small/indicator')
class Indicator(Resource):
    @ns.doc(description='根据小流域参数计算多个指标')
    @ns.expect(indicator_input)
    @ns.marshal_with(indicator_output)
    def post(self):
        """计算洪峰模数、汇流时间、峰值时间等指标

        禾水小流域
        {
          "P": 120,
          "F": 10,
          "A": 256.8,
          "k": 0.25,
          "J": 0.012,
          "S": 15.2,
          "n": 0.045,
          "f": 1.5,
          "L": 28.6,
          "RH": 0.2,
          "Ks": 0.38
        }

        蜀水小流域基础属性表
        {
          "P": 150,
          "F": 10,
          "A": 1287,
          "k": 0.25,
          "J": 0.025,
          "S": 18.5,
          "n": 0.055,
          "f": 2.0,
          "L": 65,
          "RH": 0.25,
          "Ks": 3.28
        }

        """
        data = api.payload
        try:
            P = data['P']
            F = data['F']
            A = data['A']
            k = data['k']
            J = data['J']
            S = data['S']
            n = data['n']
            f = data['f']
            L = data['L']
            RH = data['RH']
            Ks = data['Ks']

            S = np.tan(np.radians(S))  # 坡度（度）转坡比
            tc = confluence_time_wave(L, RH, J, n)
            qp = peak_modulus(P, F, A)
            tp = peak_time_nash(tc, Ks, P, A)
            result = {
                'peak_modulus_scs': round(qp, 3),
                'peak_modulus_stat': round(peak_modulus_stat(k, J * 100, S * 100, n, f, A), 3),
                'confluence_time_wave': round(tc / 3600, 1),
                'confluence_time_empirical': round(empirical_confluence_time(L, J) / 60, 1),
                'peak_time_nash': round(tp / 3600, 1),
                'max_confluence_time': round(max_confluence_time(tp, A, qp) / 3600, 1),
                'max_confluence_time_stat': round(max_confluence_time_stat(A, Ks, n), 1),
            }
            return result
        except Exception as e:
            ns.abort(500, f'计算失败: {str(e)}')
