#### 塔吊进程自启动

`go`程序中设置自动启动，需要指定工作目录，不然可能出现读取不到相对路径的配置文件问题。

```golang
// SetRegistry 将进程写入注册表
func SetRegistry() {
	executable, err := os.Executable()
	if err != nil {
		fmt.Println("[init]Error:", err)
		return
	}
	// 设置注册表路径
	keyPath := `SOFTWARE\Microsoft\Windows\CurrentVersion\Run`
	// 检查操作系统是否为 Windows
	if runtime.GOOS == "windows" {
		// 获取可执行文件的文件名
		exeName := strings.TrimSuffix(filepath.Base(executable), ".exe")
		//fmt.Println("可执行文件名为:", exeName)
		// 打开注册表项
		k, err1 := registry.OpenKey(registry.CURRENT_USER, keyPath, registry.ALL_ACCESS)
		if err1 != nil {
			fmt.Println("Error opening registry key:", err1)
			return
		}
		defer k.Close()

		// 检查键是否存在
		_, _, err = k.GetStringValue(exeName)
		if err == nil {
			fmt.Println("Registry key already exists.")
			// 删除已有的注册表键
			if err := k.DeleteValue(exeName); err != nil {
				fmt.Println("Error deleting existing registry key:", err)
				return
			}
		} else if err != registry.ErrNotExist {
			fmt.Println("Error checking registry key:", err)
			return
		}
		//fmt.Println("注册表中不存在")
		// 设置注册表项
		exeDir := filepath.Dir(executable)
		regValue := fmt.Sprintf(`cmd /c "cd /d %s && %s"`, exeDir, executable)
		if err2 := k.SetStringValue(exeName, regValue); err2 != nil {
			fmt.Println("Error setting registry key:", err2)
			return
		}
		fmt.Println("Added to registry startup.")
	} else {
		fmt.Println("This platform is not supported.")
	}
}
```
